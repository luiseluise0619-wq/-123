# SQLAlchemy PostgreSQL Schema Models (models.py)

class Project(Base):
    __tablename__ = "projects"
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    name = Column(String, nullable=False)
    target_type = Column(String) # 'url', 'github', 'code', 'openapi'
    target_url = Column(String, nullable=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    audits = relationship("Audit", back_populates="project")

class Audit(Base):
    __tablename__ = "audits"
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    project_id = Column(UUID, ForeignKey("projects.id"))
    scan_mode = Column(String) # 'standard', 'hardcore'
    status = Column(String) # 'running', 'completed', 'failed'
    security_score = Column(Integer)
    critical_count = Column(Integer)
    high_count = Column(Integer)
    medium_count = Column(Integer)
    low_count = Column(Integer)
    created_at = Column(DateTime, default=datetime.utcnow)

class Vulnerability(Base):
    __tablename__ = "vulnerabilities"
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    audit_id = Column(UUID, ForeignKey("audits.id"))
    title = Column(String, nullable=False)
    severity = Column(String) # 'Critical', 'High', 'Medium', 'Low'
    cve_id = Column(String, nullable=True)
    location = Column(String)
    description = Column(Text)
    attack_scenario = Column(Text)
    vulnerable_code = Column(Text)
    fixed_code = Column(Text)
    remediation_reason = Column(Text)
    is_resolved = Column(Boolean, default=False)
